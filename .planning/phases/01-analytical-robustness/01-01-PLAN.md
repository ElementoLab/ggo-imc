```python
class TestSilhouetteAnalysis:
    def test_silhouette_scores_are_numeric(self):
        from sklearn.metrics import silhouette_score
        from sklearn.cluster import AgglomerativeClustering
        import numpy as np
        X = np.random.rand(30, 5)
        scores = []
        for k in range(2, 7):
            labels = AgglomerativeClustering(n_clusters=k, linkage='ward').fit_predict(X)
            scores.append(silhouette_score(X, labels))
        assert len(scores) == 5
        assert all(isinstance(s, float) for s in scores)
        assert all(-1 <= s <= 1 for s in scores)

class TestKmeansConcordance:
    def test_ari_in_range(self):
        from sklearn.metrics import adjusted_rand_score
        import numpy as np
        a = np.array([0, 0, 1, 1, 2, 2])
        b = np.array([0, 0, 1, 1, 2, 2])
        ari = adjusted_rand_score(a, b)
        assert -0.5 <= ari <= 1.0

    def test_percent_agreement_perfect(self):
        from sklearn.metrics import confusion_matrix
        from scipy.optimize import linear_sum_assignment
        import numpy as np
        a = np.array([0, 1, 2, 3, 0, 1])
        cm = confusion_matrix(a, a)
        row_ind, col_ind = linear_sum_assignment(-cm)
        pct = cm[row_ind, col_ind].sum() / len(a)
        assert pct == 1.0

class TestG4SensSpec:
    def test_sensitivity_specificity_in_range(self):
        from statsmodels.stats.proportion import proportion_confint
        # 6 out of 8 positives correctly identified
        sens = 6 / 8
        ci = proportion_confint(6, 8, method='wilson')
        assert 0 <= sens <= 1
        assert ci[0] < ci[1]
        assert ci[0] >= 0 and ci[1] <= 1
```
