```python
def covariate_bar_plot(obs_df, group_col, covariate_col, ax, title):
    """Stacked bar plot of covariate % per group with chi2/Fisher p-value annotation."""
    ct = pd.crosstab(obs_df[group_col], obs_df[covariate_col])
    # Fisher's exact only for 2x2; use chi2 for larger tables
    if ct.shape == (2, 2) and (ct.values < 5).any():
        _, pval = fisher_exact(ct.values)
    else:
        _, pval, _, _ = chi2_contingency(ct.values)
    ct_pct = ct.div(ct.sum(axis=1), axis=0) * 100
    ct_pct.plot(kind='bar', stacked=True, ax=ax, legend=True)
    ax.set_title(f'{title}\np = {pval:.3g}', fontsize=9)
    ax.set_xlabel('Group')
    ax.set_ylabel('Proportion (%)')
    ax.tick_params(axis='x', rotation=0)
    return pval
```

class TestCovariateBreakdown:
    def test_crosstab_shape(self):
        obs = pd.DataFrame({
            "Group": ["G1", "G1", "G2", "G2", "G3", "G3"],
            "EGFR": ["WT", "MUT", "WT", "WT", "MUT", "MUT"],
        })
        ct = pd.crosstab(obs["Group"], obs["EGFR"])
        assert ct.shape[0] == 3  # 3 groups
        assert ct.shape[1] == 2  # WT, MUT

    def test_chi2_returns_float_pval(self):
        from scipy.stats import chi2_contingency
        import numpy as np
        ct = np.array([[10, 5], [8, 7], [6, 9]])
        _, pval, _, _ = chi2_contingency(ct)
        assert isinstance(pval, float)
        assert 0 <= pval <= 1

class TestIASLCMapping:
    def test_mm_class_values_map_cleanly(self):
        mm_map = {'ais': 'AIS (non-invasive)', 'mia': 'MIA', 'invasive': 'IAC (invasive)'}
        raw_values = ['ais', 'mia', 'invasive', 'ais', 'mia']
        mapped = [mm_map[v] for v in raw_values]
        assert len(mapped) == 5
        assert all(v in mm_map.values() for v in mapped)

class TestPNSIACCount:
    def test_pns_iac_filter(self):

        positive = (obs["radio"] == "PNS") & (obs["pathology"] == "IAC")
        assert positive.sum() == 2

        in_g4 = (obs["Group"] == "G4") & positive
